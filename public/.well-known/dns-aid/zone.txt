; DNS-AID (DNS for AI Discovery) zone fragment for dreyv.com
; Spec: https://datatracker.ietf.org/doc/draft-mozleywilliams-dnsop-dnsaid/
;
; Each capability is published as a pair:
;   SVCB — transport (alpn, port) per RFC 9460 registered SvcParamKeys
;   TXT  — endpoint=<path> companion (since `endpoint` is not a registered
;          SvcParamKey, Cloudflare and other validating resolvers reject it
;          inside SVCB; we carry it alongside instead).
;
; Publish under the apex zone and sign with DNSSEC.

$ORIGIN dreyv.com.

; Top-level agent discovery index
_index._agents   3600 IN SVCB 1 dreyv.com. ( alpn="h2,http/1.1" port=443 )
_index._agents   3600 IN TXT  "endpoint=/.well-known/agent-skills/index.json"

; MCP Streamable HTTP
_mcp._agents     3600 IN SVCB 1 dreyv.com. ( alpn="h2,http/1.1" port=443 )
_mcp._agents     3600 IN TXT  "endpoint=/api/mcp"

; ACP commerce discovery
_acp._agents     3600 IN SVCB 1 dreyv.com. ( alpn="h2,http/1.1" port=443 )
_acp._agents     3600 IN TXT  "endpoint=/.well-known/acp.json"

; A2A — agent-to-agent surface
_a2a._agents     3600 IN SVCB 1 dreyv.com. ( alpn="h2,http/1.1" port=443 )
_a2a._agents     3600 IN TXT  "endpoint=/api/mcp"

; OAuth / OIDC discovery
_oauth._agents   3600 IN SVCB 1 dreyv.com. ( alpn="h2,http/1.1" port=443 )
_oauth._agents   3600 IN TXT  "endpoint=/.well-known/openid-configuration"

; x402 payment capability
_x402._agents    3600 IN SVCB 1 dreyv.com. ( alpn="h2,http/1.1" port=443 )
_x402._agents    3600 IN TXT  "endpoint=/.well-known/x402"
